\begin{equation}
\label{eq:{{LABEL_SLUG}}-example}
\max_{x\in\mathcal{X}}\quad \sum_{t\in\mathcal{T}}\lambda_t\,x_t\,\dt \;-\; c\sum_{t\in\mathcal{T}} x_t\,\dt
\qquad\text{s.t.}\qquad 0\le x_t\le \bar{x}\quad\forall t .
\end{equation}
